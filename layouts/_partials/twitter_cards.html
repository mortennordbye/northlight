{{- /* Twitter/X card tags. Shadows Hugo's embedded twitter_cards.html so that the image lookup matches opengraph.html exactly — the embedded pair can disagree about which image a page has, and a card that shows a different picture than the OpenGraph preview is the kind of bug nobody notices until someone shares the link. summary_large_image when there is a cover, summary when there is not: a large-image card with no image renders as an empty box. */ -}} {{- $desc := "" -}} {{- with .Description }}{{ $desc = . }} {{- else }}{{ with .Summary }}{{ $desc = . | plainify | chomp }} {{- else }}{{ $desc = site.Params.description }}{{ end }}{{ end -}} {{- $image := "" -}} {{- with .Resources.GetMatch "{cover,featured}.*" }}{{ $image = .Permalink }}{{ end -}} {{- /* A page with no cover of its own falls back to the site-wide social image, so a link to it still previews as a card rather than as bare text. */ -}} {{- if not $image }}{{ with partial "image-url.html" (partialCached "init.html" .).defaultSocialImage }}{{ $image = . | absURL }}{{ end }}{{ end -}} {{- if not $image }}{{ with site.Params.image }}{{ $image = . | absURL }}{{ end }}{{ end -}} {{- with $desc }} {{- end }} {{- with $image }} {{- end }}
