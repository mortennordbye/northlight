{{- /* OpenGraph. This shadows Hugo's embedded opengraph.html deliberately. The embedded one renders article:tag from .LinkTitle, which is Hugo's title-cased form of a term — so `network-policy` goes out as `Network-Policy`. Every other tag in this theme renders as the author wrote it, and a social card that disagrees with the page is a small lie. .Data.Term is the raw term. It is also narrower than the embedded template on purpose: no author facebook or series handling, because this theme has neither. Keep it that way — the reason to own this file is control, and control is only worth having while the file stays small enough to read. */ -}} {{- $title := cond .IsHome site.Title .Title -}} {{- $desc := "" -}} {{- with .Description }}{{ $desc = . }} {{- else }}{{ with .Summary }}{{ $desc = . | plainify | chomp }} {{- else }}{{ $desc = site.Params.description }}{{ end }}{{ end -}} {{- with $desc }} {{- end }} {{- if and .IsPage (in site.Params.mainSections .Section) }} {{- with .Section }} {{- end }} {{- if not .Date.IsZero }} {{- end }} {{- if not .Lastmod.IsZero }} {{- end }} {{- range .GetTerms "tags" }} {{- end }} {{- end }} {{- /* The cover, as an absolute URL — a relative og:image is ignored by every consumer of it. Falls back to a site-wide image if a page has no cover. */ -}} {{- $image := "" -}} {{- with .Resources.GetMatch "{cover,featured}.*" }}{{ $image = .Permalink }}{{ end -}} {{- /* A page with no cover of its own falls back to the site-wide social image, so a link to it still previews as a card rather than as bare text. */ -}} {{- if not $image }}{{ with partial "image-url.html" (partialCached "init.html" .).defaultSocialImage }}{{ $image = . | absURL }}{{ end }}{{ end -}} {{- if not $image }}{{ with site.Params.image }}{{ $image = . | absURL }}{{ end }}{{ end -}} {{- with $image }} {{- end }}
