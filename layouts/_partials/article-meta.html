{{- $cfg := partialCached "init.html" . -}} {{- $a := $cfg.article -}} {{- $author := $cfg.author -}}
{{- $authors := partial "authors.html" . -}} {{- if and $a.showAuthor $authors }} {{- /* Avatars are sized and cropped in CSS rather than by Hugo, so an SVG works as well as a raster one. Supply a square image, 160px or larger. Resolved through image-url.html so a configured-but-missing file renders nothing instead of a 404 on every page. With several authors the avatars stack into one group and the names run as a list, rather than repeating an avatar-plus-name pair per person — three of those in a meta line reads as three separate bylines. */ -}} {{- $avatars := slice -}} {{- range $authors }}{{ with partial "image-url.html" .image }}{{ $avatars = $avatars | append . }}{{ end }}{{ end -}} {{- with $avatars }} {{- range . }}{{ end }} {{- end }} {{- range $i, $au := $authors -}} {{- if $i }}{{ i18n "authorSeparator" }}{{ end -}} {{- if and $a.showAuthorsBadges $au.term (index site.Taxonomies "authors") -}} {{- with site.GetPage (printf "/authors/%s" ($au.term | urlize)) -}} {{ $au.name }} {{- else -}} {{ $au.name }} {{- end -}} {{- else -}} {{ $au.name }} {{- end -}} {{- end -}} {{- end }}
{{- $items := slice -}} {{- if and $a.showDate (not .Date.IsZero) -}} {{- $items = $items | append (printf `%s` (.Date.Format "2006-01-02") (.Date.Format $cfg.dateFormat)) -}} {{- end -}} {{- /* The updated date is only worth showing when it says something the publication date does not. Hugo defaults Lastmod to Date, so without a comparison every post would claim to have been updated on the day it was written. The comparison is on the *formatted* dates rather than the timestamps, because that is what the reader sees. A post published in the morning and corrected that afternoon has a later Lastmod to the second, but both render as the same day, and "27 Jul 2026 · Updated 27 Jul 2026" is noise. Whether it changed is Git's question; whether it looks changed is this one. */ -}} {{- $shown := .Date.Format $cfg.dateFormat -}} {{- $updatedShown := .Lastmod.Format $cfg.dateFormat -}} {{- if and $a.showDateUpdated (not .Lastmod.IsZero) (gt .Lastmod.Unix .Date.Unix) (ne $updatedShown $shown) -}} {{- $stamp := printf `%s` (.Lastmod.Format "2006-01-02") (.Lastmod.Format $cfg.dateFormat) -}} {{- $items = $items | append (i18n "updatedOn" ($stamp | safeHTML)) -}} {{- end -}} {{- if $a.showReadingTime -}} {{- $items = $items | append (i18n "readingTime" .ReadingTime) -}} {{- end -}} {{- if $a.showWordCount -}} {{- $items = $items | append (i18n "wordCount" (dict "Count" .WordCount "Formatted" (lang.FormatNumber 0 .WordCount))) -}} {{- end -}} {{- range $i, $item := $items }} {{- if $i }}{{ end }} {{ $item | safeHTML }} {{- end }}
{{- /* Edit link. Only rendered when a base URL is configured, because a link that goes nowhere is worse than no link. editAppendPath is on by default, so the configured value is the repository's content root rather than a per-post URL, which is what nearly every site wants. */ -}} {{- if and $a.showEdit $a.editURL }} {{- $url := $a.editURL -}} {{- if and $a.editAppendPath .File }} {{- $url = printf "%s/%s" (strings.TrimSuffix "/" $a.editURL) .File.Path -}} {{- end }} {{ partialCached "icon.html" "pencil" "pencil" }}{{ i18n "editThisPage" }} {{- end }}
