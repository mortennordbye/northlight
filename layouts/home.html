{{ define "main" }} {{- /* Home page dispatcher, and nothing else. It resolves the layout, gathers the post list once, and hands both to the chosen partial in `_partials/home/`. Keeping the gathering here is the point: ten layouts would otherwise grow ten copies of it, drift apart, and each need fixing separately. It is also what makes `custom` possible without forking the theme. `home.layout` defaults to `stack`, which is the arrangement this theme shipped with, so a site that never sets the param sees no change. */ -}} {{- $cfg := partialCached "init.html" . -}} {{- $home := site.Params.home | default dict -}} {{- $layout := lower ($home.layout | default "stack") -}} {{- $known := slice "stack" "page" "profile" "hero" "card" "background" "split" "gallery" "archive" "custom" -}} {{- if not (in $known $layout) -}} {{- /* Fail rather than falling back to stack. A typo would otherwise render the default homepage and look like the param had no effect, which is the kind of thing someone debugs for an hour. */ -}} {{- errorf "home.layout: unknown layout %q — expected one of %s" $layout (delimit $known ", ") -}} {{- end -}} {{- /* One gather, shared by every layout. Sorted by Hugo's default, newest first. */ -}} {{- $posts := where site.RegularPages "Section" "in" site.Params.mainSections -}} {{- partial (printf "home/%s.html" $layout) (dict "ctx" . "cfg" $cfg "home" $home "posts" $posts "author" $cfg.author ) -}} {{ end }}
